```ts
import type { StoryReactFlowNode, StoryReactFlowEdge } from '@/types/graph.types';

export interface StoryAdjacency {
  children: Map<string, string[]>;
  parents: Map<string, string>;
  roots: string[];
}

export function buildStoryGraph(
  nodes: StoryReactFlowNode[],
  edges: StoryReactFlowEdge[],
): StoryAdjacency {
  const storyIds = new Set(
    nodes.filter((n) => n.data.type === 'story').map((n) => n.id),
  );
  const choiceIds = new Set(
    nodes.filter((n) => n.data.type === 'choice').map((n) => n.id),
  );

  // Build adjacency from raw edges
  const adj = new Map<string, string[]>();
  for (const e of edges) {
    if (!adj.has(e.source)) adj.set(e.source, []);
    adj.get(e.source)!.push(e.target);
  }

  // Traverse story→choice→story chains to build story-only links
  const children = new Map<string, string[]>();
  const parents = new Map<string, string>();

  for (const storyId of storyIds) {
    const choiceChildren = (adj.get(storyId) ?? []).filter((id) => choiceIds.has(id));
    for (const choiceId of choiceChildren) {
      const storyGrandchildren = (adj.get(choiceId) ?? []).filter((id) => storyIds.has(id));
      for (const childStory of storyGrandchildren) {
        if (!children.has(storyId)) children.set(storyId, []);
        children.get(storyId)!.push(childStory);
        parents.set(childStory, storyId);
      }
    }
  }

  const roots = [...storyIds].filter((id) => !parents.has(id));

  return { children, parents, roots };
}
```

```ts
export function computeLeafDistances(
  graph: StoryAdjacency,
): Map<string, Set<number>> {
  const distances = new Map<string, Set<number>>();

  // All story node IDs = roots + all children values + all parent keys
  const allStoryIds = new Set<string>([
    ...graph.roots,
    ...graph.parents.keys(),
    ...[...graph.children.values()].flat(),
  ]);

  // Identify leaves: story nodes with no children
  const leaves = new Set(
    [...allStoryIds].filter((id) => !graph.children.has(id)),
  );

  // For each node, DFS to compute distances to all descendant leaves
  function dfs(nodeId: string): Set<number> {
    if (distances.has(nodeId)) return distances.get(nodeId)!;

    const result = new Set<number>();
    if (leaves.has(nodeId)) {
      result.add(0);
    }

    const kids = graph.children.get(nodeId) ?? [];
    for (const child of kids) {
      const childDists = dfs(child);
      for (const d of childDists) {
        result.add(d + 1);
      }
    }

    distances.set(nodeId, result);
    return result;
  }

  for (const root of graph.roots) {
    dfs(root);
  }

  return distances;
}

export function getRowNodes(
  leafDistances: Map<string, Set<number>>,
  k: number,
): string[] {
  return [...leafDistances.entries()]
    .filter(([, dists]) => dists.has(k))
    .map(([id]) => id);
}
```

```ts
export function dfsOrder(
  rowNodeIds: string[],
  graph: StoryAdjacency,
): string[] {
  if (rowNodeIds.length === 0) return [];

  const rowSet = new Set(rowNodeIds);
  const result: string[] = [];
  const visited = new Set<string>();

  function dfs(nodeId: string) {
    if (visited.has(nodeId)) return;
    visited.add(nodeId);

    if (rowSet.has(nodeId)) {
      result.push(nodeId);
    }

    const kids = graph.children.get(nodeId) ?? [];
    for (const child of kids) {
      dfs(child);
    }
  }

  for (const root of graph.roots) {
    dfs(root);
  }

  return result;
}
```

```ts
export interface RowLayoutResult {
  orderedNodeIds: string[];
  distances: Map<string, Map<string, number>>;
  maxDepth: number;
}

export function computeGraphDistances(
  rowNodeIds: string[],
  graph: StoryAdjacency,
): Map<string, Map<string, number>> {
  // Build undirected adjacency from the story graph
  const undirected = new Map<string, string[]>();
  const allStoryIds = new Set<string>([
    ...graph.roots,
    ...graph.parents.keys(),
    ...[...graph.children.values()].flat(),
  ]);

  for (const id of allStoryIds) {
    undirected.set(id, []);
  }
  for (const [parent, kids] of graph.children) {
    for (const kid of kids) {
      undirected.get(parent)!.push(kid);
      undirected.get(kid)!.push(parent);
    }
  }

  const rowSet = new Set(rowNodeIds);
  const result = new Map<string, Map<string, number>>();

  // BFS from each row node to compute distances to all other row nodes
  for (const startId of rowNodeIds) {
    const distMap = new Map<string, number>();
    distMap.set(startId, 0);
    const queue = [startId];
    const visited = new Set([startId]);

    while (queue.length > 0) {
      const current = queue.shift()!;
      const currentDist = distMap.get(current)!;
      for (const neighbor of undirected.get(current) ?? []) {
        if (!visited.has(neighbor)) {
          visited.add(neighbor);
          distMap.set(neighbor, currentDist + 1);
          queue.push(neighbor);
        }
      }
    }

    // Filter to only row nodes
    const rowDistMap = new Map<string, number>();
    for (const id of rowNodeIds) {
      rowDistMap.set(id, distMap.get(id) ?? Infinity);
    }
    result.set(startId, rowDistMap);
  }

  return result;
}

export function computeRowLayout(
  nodes: StoryReactFlowNode[],
  edges: StoryReactFlowEdge[],
  rowDepth: number,
): RowLayoutResult {
  const graph = buildStoryGraph(nodes, edges);
  const leafDistances = computeLeafDistances(graph);

  // Compute maxDepth
  let maxDepth = 0;
  for (const dists of leafDistances.values()) {
    for (const d of dists) {
      if (d > maxDepth) maxDepth = d;
    }
  }

  const rowNodeIds = getRowNodes(leafDistances, rowDepth);
  const orderedNodeIds = dfsOrder(rowNodeIds, graph);
  const distances = computeGraphDistances(orderedNodeIds, graph);

  return { orderedNodeIds, distances, maxDepth };
}
```

```ts
import { useReactFlow, type Viewport } from 'reactflow';
import type { TransformedGraph } from '@/utils/graphTransform';
import type { StoryReactFlowNode, StoryReactFlowEdge } from '@/types/graph.types';
import { GRAPH_VISUAL_CONFIG } from '@/config/graph.config';
import { computeRowLayout, type RowLayoutResult } from '@/utils/rowLayoutEngine';

const ROW = GRAPH_VISUAL_CONFIG.rowMode;
const CHOICE_Y = ROW.storyY + GRAPH_VISUAL_CONFIG.storyNode.height + ROW.choiceGap;

export interface UseRowLayoutResult {
  nodes: StoryReactFlowNode[];
  edges: StoryReactFlowEdge[];
  maxDepth: number;
  centeredNodeId: string | null;
  onViewportChange: (viewport: Viewport) => void;
}

export function useRowLayout(
  graph: TransformedGraph | null,
  rowDepth: number,
  containerWidth: number,
): UseRowLayoutResult {
  const reactFlow = useReactFlow();

  // Compute row layout from engine
  const engineResult = useMemo<RowLayoutResult | null>(() => {
    if (!graph || !graph.nodes.length) return null;
    return computeRowLayout(graph.nodes, graph.edges, rowDepth);
  }, [graph, rowDepth]);

  // Build position map for story nodes
  const storyPositions = useMemo(() => {
    const map = new Map<string, { x: number; y: number }>();
    if (!engineResult) return map;
    engineResult.orderedNodeIds.forEach((id, index) => {
      map.set(id, { x: index * ROW.spacing, y: ROW.storyY });
    });
    return map;
  }, [engineResult]);

  // Center detection — initialize synchronously to avoid empty-content flash
  const initialCenteredId = useMemo(() => {
    if (!engineResult || !engineResult.orderedNodeIds.length) return null;
    const rowSet = new Set(engineResult.orderedNodeIds);
    if (graph?.latestStoryNodeId && rowSet.has(graph.latestStoryNodeId)) {
      return graph.latestStoryNodeId;
    }
    return engineResult.orderedNodeIds[0];
  }, [engineResult, graph?.latestStoryNodeId]);

  const [centeredNodeId, setCenteredNodeId] = useState<string | null>(initialCenteredId);
  const centeredRef = useRef<string | null>(initialCenteredId);

  // Snap viewport when initial centered node changes (depth or graph change)
  useEffect(() => {
    if (!initialCenteredId) return;
    setCenteredNodeId(initialCenteredId);
    centeredRef.current = initialCenteredId;

    // Snap viewport to this node
    const pos = storyPositions.get(latestInRow);
    if (pos) {
      requestAnimationFrame(() => {
        reactFlow.setCenter(pos.x, ROW.storyY, { zoom: 1, duration: 300 });
      });
    }
  }, [initialCenteredId]); // eslint-disable-line react-hooks/exhaustive-deps

  // Viewport change handler — detect new centered node
  const onViewportChange = useCallback(
    (viewport: Viewport) => {
      if (!engineResult || !engineResult.orderedNodeIds.length) return;

      const centerX = (-viewport.x + containerWidth / 2) / viewport.zoom;
      let closestId = engineResult.orderedNodeIds[0];
      let closestDist = Infinity;

      for (const id of engineResult.orderedNodeIds) {
        const pos = storyPositions.get(id);
        if (!pos) continue;
        const dist = Math.abs(pos.x - centerX);
        if (dist < closestDist) {
          closestDist = dist;
          closestId = id;
        }
      }

      if (closestId !== centeredRef.current) {
        centeredRef.current = closestId;
        setCenteredNodeId(closestId);
      }
    },
    [engineResult, storyPositions, containerWidth],
  );

  // Build final nodes and edges
  const { nodes, edges } = useMemo(() => {
    if (!graph || !engineResult || !centeredNodeId) {
      return { nodes: [] as StoryReactFlowNode[], edges: [] as StoryReactFlowEdge[] };
    }

    const orderedIds = engineResult.orderedNodeIds;
    const centerIdx = orderedIds.indexOf(centeredNodeId);
    if (centerIdx === -1) {
      return { nodes: [] as StoryReactFlowNode[], edges: [] as StoryReactFlowEdge[] };
    }

    // Visible story window: center ± 2
    const visibleStart = Math.max(0, centerIdx - 2);
    const visibleEnd = Math.min(orderedIds.length - 1, centerIdx + 2);
    const visibleStoryIds = orderedIds.slice(visibleStart, visibleEnd + 1);
    const visibleStorySet = new Set(visibleStoryIds);

    // Choice window: center ± 1
    const choiceStart = Math.max(0, centerIdx - 1);
    const choiceEnd = Math.min(orderedIds.length - 1, centerIdx + 1);
    const choiceStoryIds = new Set(orderedIds.slice(choiceStart, choiceEnd + 1));

    // Distances from centered node
    const distMap = engineResult.distances.get(centeredNodeId);

    // Build story nodes with positions and distanceFromCenter
    const resultNodes: StoryReactFlowNode[] = [];
    const resultEdges: StoryReactFlowEdge[] = [];

    const originalNodeMap = new Map(graph.nodes.map((n) => [n.id, n]));

    for (const storyId of visibleStoryIds) {
      const original = originalNodeMap.get(storyId);
      if (!original) continue;
      const pos = storyPositions.get(storyId)!;
      const graphDist = distMap?.get(storyId) ?? 0;

      resultNodes.push({
        ...original,
        position: pos,
        data: { ...original.data, distanceFromCenter: graphDist },
      });

      // Add choice nodes if in choice window
      if (choiceStoryIds.has(storyId)) {
        // Find choice children of this story node from original edges
        const choiceEdges = graph.edges.filter((e) => e.source === storyId);
        const choiceIds = choiceEdges.map((e) => e.target);
        const validChoices = choiceIds.filter((id) => {
          const node = originalNodeMap.get(id);
          return node && node.data.type === 'choice';
        });

        const choiceCount = validChoices.length;
        const totalChoiceWidth = choiceCount * GRAPH_VISUAL_CONFIG.choiceNode.width;
        const choiceGap = 20;
        const totalWidth = totalChoiceWidth + (choiceCount - 1) * choiceGap;
        const startX = pos.x + GRAPH_VISUAL_CONFIG.storyNode.width / 2 - totalWidth / 2;

        validChoices.forEach((choiceId, i) => {
          const choiceOriginal = originalNodeMap.get(choiceId)!;
          const choiceX = startX + i * (GRAPH_VISUAL_CONFIG.choiceNode.width + choiceGap);

          resultNodes.push({
            ...choiceOriginal,
            position: { x: choiceX, y: CHOICE_Y },
            data: { ...choiceOriginal.data, distanceFromCenter: graphDist },
          });

          // Edge from story to choice
          resultEdges.push({
            id: `row-${storyId}-${choiceId}`,
            source: storyId,
            target: choiceId,
            type: 'smoothstep',
          });
        });
      }
    }

    return { nodes: resultNodes, edges: resultEdges };
  }, [graph, engineResult, centeredNodeId, storyPositions]);

  return {
    nodes,
    edges,
    maxDepth: engineResult?.maxDepth ?? 0,
    centeredNodeId,
    onViewportChange,
  };
}
```
